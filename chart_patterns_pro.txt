// PIVOT POINT DETECTION
// Based on zeta-zetra pivot_points.py - find_pivot_point()
// ═══════════════════════════════════════════════════════════════
f_is_pivot_low(int idx, int left, int right) =>
    bool is_low = true
    for i = 1 to left
        if low[idx] > low[idx + i]
            is_low := false
            break
    if is_low
        for i = 1 to right
            if idx - i >= 0
                if low[idx] > low[idx - i]
                    is_low := false
                    break
    is_low

f_is_pivot_high(int idx, int left, int right) =>
    bool is_high = true
    for i = 1 to left
        if high[idx] < high[idx + i]
            is_high := false
            break
    if is_high
        for i = 1 to right
            if idx - i >= 0
                if high[idx] < high[idx - i]
                    is_high := false
                    break
    is_high

// Returns: 0=none, 1=pivot low, 2=pivot high, 3=both
f_pivot_type(int idx, int left, int right) =>
    bool pl = f_is_pivot_low(idx, left, right)
    bool ph = f_is_pivot_high(idx, left, right)
    pl and ph ? 3 : pl ? 1 : ph ? 2 : 0

// ═══════════════════════════════════════════════════════════════
// LINEAR REGRESSION (replaces scipy.stats.linregress)
// Returns [slope, intercept, r_squared]
// ═══════════════════════════════════════════════════════════════
f_linregress(array<float> x_arr, array<float> y_arr) =>
    int n = array.size(x_arr)
    float slope = 0.0
    float intercept = 0.0
    float r2 = 0.0

    if n >= 2
        float sum_x = 0.0
        float sum_y = 0.0
        float sum_xy = 0.0
        float sum_x2 = 0.0
        float sum_y2 = 0.0

        for i = 0 to n - 1
            float xi = array.get(x_arr, i)
            float yi = array.get(y_arr, i)
            sum_x  += xi
            sum_y  += yi
            sum_xy += xi * yi
            sum_x2 += xi * xi
            sum_y2 += yi * yi

        float mean_x = sum_x / n
        float mean_y = sum_y / n
        float ss_xx = sum_x2 - n * mean_x * mean_x
        float ss_yy = sum_y2 - n * mean_y * mean_y
        float ss_xy = sum_xy - n * mean_x * mean_y

        if math.abs(ss_xx) > 1e-10
            slope     := ss_xy / ss_xx
            intercept := mean_y - slope * mean_x

            if math.abs(ss_yy) > 1e-10
                r2 := (ss_xy * ss_xy) / (ss_xx * ss_yy)

    [slope, intercept, r2]

// HELPER: Collect pivots in a lookback window
// ═══════════════════════════════════════════════════════════════
f_collect_pivots(int lookback, int left, int right) =>
    array<int>   hi_bars   = array.new<int>()
    array<float> hi_prices = array.new<float>()
    array<int>   lo_bars   = array.new<int>()
    array<float> lo_prices = array.new<float>()
    array<int>   all_bars  = array.new<int>()
    array<float> all_pos   = array.new<float>()
    array<int>   all_types = array.new<int>()

    int start = math.min(lookback, bar_index)
    for i = start to 0
        int pt = f_pivot_type(i, left, right)
        if pt == 1 or pt == 3
            array.push(lo_bars, bar_index - i)
            array.push(lo_prices, low[i])
        if pt == 2 or pt == 3
            array.push(hi_bars, bar_index - i)
            array.push(hi_prices, high[i])
        if pt != 0
            array.push(all_bars, bar_index - i)
            array.push(all_pos, pt == 1 ? low[i] : high[i])
            array.push(all_types, pt)

    [hi_bars, hi_prices, lo_bars, lo_prices, all_bars, all_pos, all_types]

// PATTERN DETECTION: DOUBLE TOP / BOTTOM
// Based on zeta-zetra doubles.py
// Requires exactly 5 pivots in window forming W or M shape
// ═══════════════════════════════════════════════════════════════
f_detect_double(int lookback, int left, int right) =>
    PatternResult result = na

    if not i_double_en
        result
    else
        // Collect all pivots
        array<int>   pvt_bars = array.new<int>()
        array<float> pvt_pos  = array.new<float>()

        int start = math.min(lookback, bar_index)
        for i = start to 0
            int pt = f_pivot_type(i, left, right)
            if pt == 1
                array.push(pvt_bars, bar_index - i)
                array.push(pvt_pos, low[i] - syminfo.mintick)
            else if pt == 2
                array.push(pvt_bars, bar_index - i)
                array.push(pvt_pos, high[i] + syminfo.mintick)
            else if pt == 3
                // Add as high (primary for double patterns)
                array.push(pvt_bars, bar_index - i)
                array.push(pvt_pos, high[i] + syminfo.mintick)

        if array.size(pvt_bars) == 5
            float p0 = array.get(pvt_pos, 0)
            float p1 = array.get(pvt_pos, 1)
            float p2 = array.get(pvt_pos, 2)
            float p3 = array.get(pvt_pos, 3)
            float p4 = array.get(pvt_pos, 4)

            // Double Tops: valley-PEAK-valley-PEAK-valley
            bool is_double_top = (p0 < p1) and (p0 < p3) and (p2 < p1) and (p2 < p3) and (p4 < p1) and (p4 < p3) and (p1 > p3) and (p1 / p3 <= i_double_ratio)

            // Double Bottoms: peak-TROUGH-peak-TROUGH-peak
            bool is_double_bot = (p0 > p1) and (p0 > p3) and (p2 > p1) and (p2 > p3) and (p4 > p1) and (p4 > p3) and (p1 < p3) and (p1 / p3 >= i_double_bot_ratio)

            if is_double_top
                int[] bars = array.from(array.get(pvt_bars, 0), array.get(pvt_bars, 1), array.get(pvt_bars, 2), array.get(pvt_bars, 3), array.get(pvt_bars, 4))
                float[] prices = array.from(p0, p1, p2, p3, p4)
                float neckline = math.min(p0, p2, p4)
                float peak_avg = (p1 + p3) / 2
                float tgt = neckline - (peak_avg - neckline)
                result := PatternResult.new("double_top", bar_index, bars, prices, na, na, na, na, tgt, -1)

            if is_double_bot
                int[] bars = array.from(array.get(pvt_bars, 0), array.get(pvt_bars, 1), array.get(pvt_bars, 2), array.get(pvt_bars, 3), array.get(pvt_bars, 4))
                float[] prices = array.from(p0, p1, p2, p3, p4)
                float neckline = math.max(p0, p2, p4)
                float trough_avg = (p1 + p3) / 2
                float tgt = neckline + (neckline - trough_avg)
                result := PatternResult.new("double_bottom", bar_index, bars, prices, na, na, na, na, tgt, 1)

        result

// ═══════════════════════════════════════════════════════════════
// PATTERN DETECTION: HEAD & SHOULDERS
// Based on zeta-zetra head_and_shoulders.py
// ═══════════════════════════════════════════════════════════════
f_detect_hs(int lookback) =>
    PatternResult result = na

    if not i_hs_en
        result
    else
        int half = int(lookback / 2)
        int center = half  // bars back from current bar

        // Check if center is a pivot high (main + short)
        bool main_ph = f_is_pivot_high(center, i_pivot_left_hs, i_pivot_left_hs)
        bool short_ph = f_is_pivot_high(center, i_pivot_short_hs, i_pivot_short_hs)

        if main_ph and short_ph
            // Collect short pivots around center
            array<float> maxim = array.new<float>()
            array<int>   xxmax = array.new<int>()
            array<float> minim = array.new<float>()
            array<int>   xxmin = array.new<int>()
            int maxb = 0
            int maxa = 0
            int minb = 0
            int mina = 0

            int scan_start = math.min(center + half, bar_index)
            int scan_end   = math.max(center - half, 0)

            for i = scan_start to scan_end
                bool is_spl = f_is_pivot_low(i, i_pivot_short_hs, i_pivot_short_hs)
                bool is_sph = f_is_pivot_high(i, i_pivot_short_hs, i_pivot_short_hs)

                if is_spl
                    array.push(minim, low[i])
                    array.push(xxmin, bar_index - i)
                    if i > center
                        minb += 1
                    else if i < center
                        mina += 1

                if is_sph
                    array.push(maxim, high[i])
                    array.push(xxmax, bar_index - i)
                    if i > center
                        maxb += 1
                    else if i < center
                        maxa += 1

            if minb >= 1 and mina >= 1 and maxb >= 1 and maxa >= 1 and array.size(maxim) >= 3 and array.size(minim) >= 2
                // Find head (highest peak)
                int headidx = 0
                float headval = array.get(maxim, 0)
                for i = 1 to array.size(maxim) - 1
                    if array.get(maxim, i) > headval
                        headval := array.get(maxim, i)
                        headidx := i

                // Head must not be the last
                if headidx > 0 and headidx < array.size(maxim) - 1
                    float left_shoulder  = array.get(maxim, headidx - 1)
                    float right_shoulder = array.get(maxim, headidx + 1)
                    float head_price     = array.get(maxim, headidx)

                    bool head_above_left  = head_price > left_shoulder and (head_price / left_shoulder) > i_hs_head_ratio
                    bool head_above_right = head_price > right_shoulder and (head_price / right_shoulder) > i_hs_head_ratio

                    // Neckline regression on lows
                    array<float> xxmin_f = array.new<float>()
                    for i = 0 to array.size(xxmin) - 1
                        array.push(xxmin_f, float(array.get(xxmin, i)))
                    [slmin, _, _] = f_linregress(xxmin_f, minim)

                    bool neck_flat = math.abs(slmin) <= i_hs_neck_slope

                    if head_above_left and head_above_right and neck_flat
                        int ls_bar = array.get(xxmax, headidx - 1)
                        int head_bar = array.get(xxmax, headidx)
                        int rs_bar = array.get(xxmax, headidx + 1)
                        int nl_bar = array.size(xxmin) >= 1 ? array.get(xxmin, 0) : ls_bar
                        int nr_bar = array.size(xxmin) >= 2 ? array.get(xxmin, 1) : rs_bar

                        int[] bars = array.from(ls_bar, nl_bar, head_bar, nr_bar, rs_bar)
                        float nl_price = array.size(minim) >= 1 ? array.get(minim, 0) : low[center]
                        float nr_price = array.size(minim) >= 2 ? array.get(minim, 1) : low[center]
                        float neckline = (nl_price + nr_price) / 2
                        float[] prices = array.from(left_shoulder, nl_price, head_price, nr_price, right_shoulder)
                        float tgt = neckline - (head_price - neckline)

                        result := PatternResult.new("hs", bar_index, bars, prices, na, na, na, na, tgt, -1)

        result

// ═══════════════════════════════════════════════════════════════
// PATTERN DETECTION: INVERSE HEAD & SHOULDERS
// Based on zeta-zetra inverse_head_and_shoulders.py
// ═══════════════════════════════════════════════════════════════
f_detect_ihs(int lookback) =>
    PatternResult result = na

    if not i_ihs_en
        result
    else
        int half = int(lookback / 2)
        int center = half

        bool main_pl = f_is_pivot_low(center, i_pivot_left_hs, i_pivot_left_hs)
        bool short_pl = f_is_pivot_low(center, i_pivot_short_hs, i_pivot_short_hs)

        if main_pl and short_pl
            array<float> maxim = array.new<float>()
            array<int>   xxmax = array.new<int>()
            array<float> minim = array.new<float>()
            array<int>   xxmin = array.new<int>()
            int maxb = 0
            int maxa = 0
            int minb = 0
            int mina = 0

            int scan_start = math.min(center + half, bar_index)
            int scan_end   = math.max(center - half, 0)

            for i = scan_start to scan_end
                bool is_spl = f_is_pivot_low(i, i_pivot_short_hs, i_pivot_short_hs)
                bool is_sph = f_is_pivot_high(i, i_pivot_short_hs, i_pivot_short_hs)

                if is_spl
                    array.push(minim, low[i])
                    array.push(xxmin, bar_index - i)
                    if i > center
                        minb += 1
                    else if i < center
                        mina += 1

                if is_sph
                    array.push(maxim, high[i])
                    array.push(xxmax, bar_index - i)
                    if i > center
                        maxb += 1
                    else if i < center
                        maxa += 1

            if minb >= 1 and mina >= 1 and maxb >= 1 and maxa >= 1 and array.size(minim) >= 3 and array.size(maxim) >= 2
                // Find head (lowest trough)
                int headidx = 0
                float headval = array.get(minim, 0)
                for i = 1 to array.size(minim) - 1
                    if array.get(minim, i) < headval
                        headval := array.get(minim, i)
                        headidx := i

                if headidx > 0 and headidx < array.size(minim) - 1
                    float left_shoulder  = array.get(minim, headidx - 1)
                    float right_shoulder = array.get(minim, headidx + 1)
                    float head_price     = array.get(minim, headidx)

                    float ihs_ratio = 1.0 / i_hs_head_ratio  // Invert for inverse pattern
                    bool head_below_left  = head_price < left_shoulder and (head_price / left_shoulder) < 1.0
                    bool head_below_right = head_price < right_shoulder and (head_price / right_shoulder) < 1.0

                    // Neckline regression on highs
                    array<float> xxmax_f = array.new<float>()
                    for i = 0 to array.size(xxmax) - 1
                        array.push(xxmax_f, float(array.get(xxmax, i)))
                    [slmax, _, _] = f_linregress(xxmax_f, maxim)

                    bool neck_flat = math.abs(slmax) <= i_hs_neck_slope

                    if head_below_left and head_below_right and neck_flat
                        int ls_bar = array.get(xxmin, headidx - 1)
                        int head_bar = array.get(xxmin, headidx)
                        int rs_bar = array.get(xxmin, headidx + 1)
                        int nl_bar = array.size(xxmax) >= 1 ? array.get(xxmax, 0) : ls_bar
                        int nr_bar = array.size(xxmax) >= 2 ? array.get(xxmax, 1) : rs_bar

                        int[] bars = array.from(ls_bar, nl_bar, head_bar, nr_bar, rs_bar)
                        float nl_price = array.size(maxim) >= 1 ? array.get(maxim, 0) : high[center]
                        float nr_price = array.size(maxim) >= 2 ? array.get(maxim, 1) : high[center]
                        float neckline = (nl_price + nr_price) / 2
                        float[] prices = array.from(left_shoulder, nl_price, head_price, nr_price, right_shoulder)
                        float tgt = neckline + (neckline - head_price)

                        result := PatternResult.new("ihs", bar_index, bars, prices, na, na, na, na, tgt, 1)

        result

// ═══════════════════════════════════════════════════════════════
// PATTERN DETECTION: TRIANGLES (Ascending, Descending, Symmetrical)
// Based on zeta-zetra triangles.py
// ═══════════════════════════════════════════════════════════════
f_detect_triangle(int lookback, int left, int right) =>
    PatternResult result = na

    if not i_tri_asc_en and not i_tri_desc_en and not i_tri_sym_en
        result
    else
        [hi_bars, hi_prices, lo_bars, lo_prices, _, _, _] = f_collect_pivots(lookback, left, right)

        if array.size(hi_bars) >= i_tri_min_pts and array.size(lo_bars) >= i_tri_min_pts
            // Linear regression on highs
            array<float> hi_bars_f = array.new<float>()
            for i = 0 to array.size(hi_bars) - 1
                array.push(hi_bars_f, float(array.get(hi_bars, i)))
            [slmax, intmax, r2max] = f_linregress(hi_bars_f, hi_prices)

            // Linear regression on lows
            array<float> lo_bars_f = array.new<float>()
            for i = 0 to array.size(lo_bars) - 1
                array.push(lo_bars_f, float(array.get(lo_bars, i)))
            [slmin, intmin, r2min] = f_linregress(lo_bars_f, lo_prices)

            bool good_fit = r2max >= i_tri_r2 and r2min >= i_tri_r2

            if good_fit
                // Symmetrical Triangle: highs falling, lows rising
                if i_tri_sym_en and slmin >= i_tri_flat_tol and slmax <= -i_tri_flat_tol
                    int first_bar = math.min(array.size(hi_bars) > 0 ? array.get(hi_bars, 0) : bar_index, array.size(lo_bars) > 0 ? array.get(lo_bars, 0) : bar_index)
                    int last_bar = bar_index
                    float height = (intmax + slmax * first_bar) - (intmin + slmin * first_bar)
                    // Breakout direction unknown for symmetrical
                    float tgt_up = (slmax * last_bar + intmax) + height * 0.618
                    float tgt_dn = (slmin * last_bar + intmin) - height * 0.618

                    int[] bars = array.new<int>()
                    float[] prices = array.new<float>()
                    for i = 0 to array.size(hi_bars) - 1
                        array.push(bars, array.get(hi_bars, i))
                        array.push(prices, array.get(hi_prices, i))
                    for i = 0 to array.size(lo_bars) - 1
                        array.push(bars, array.get(lo_bars, i))
                        array.push(prices, array.get(lo_prices, i))

                    result := PatternResult.new("sym_tri", bar_index, bars, prices, slmax, slmin, intmax, intmin, close > (slmax * bar_index + intmax + slmin * bar_index + intmin) / 2 ? tgt_up : tgt_dn, 0)

                // Ascending Triangle: lows rising, highs flat
                else if i_tri_asc_en and slmin >= i_tri_flat_tol and math.abs(slmax) <= i_tri_flat_tol
                    float resistance = intmax + slmax * bar_index
                    float support = intmin + slmin * bar_index
                    float height = resistance - support
                    float tgt = resistance + height * 0.618

                    int[] bars = array.new<int>()
                    float[] prices = array.new<float>()
                    for i = 0 to array.size(hi_bars) - 1
                        array.push(bars, array.get(hi_bars, i))
                        array.push(prices, array.get(hi_prices, i))
                    for i = 0 to array.size(lo_bars) - 1
                        array.push(bars, array.get(lo_bars, i))
                        array.push(prices, array.get(lo_prices, i))

                    result := PatternResult.new("asc_tri", bar_index, bars, prices, slmax, slmin, intmax, intmin, tgt, 1)

                // Descending Triangle: highs falling, lows flat
                else if i_tri_desc_en and slmax <= -i_tri_flat_tol and math.abs(slmin) <= i_tri_flat_tol
                    float resistance = intmax + slmax * bar_index
                    float support = intmin + slmin * bar_index
                    float height = resistance - support
                    float tgt = support - height * 0.618

                    int[] bars = array.new<int>()
                    float[] prices = array.new<float>()
                    for i = 0 to array.size(hi_bars) - 1
                        array.push(bars, array.get(hi_bars, i))
                        array.push(prices, array.get(hi_prices, i))
                    for i = 0 to array.size(lo_bars) - 1
                        array.push(bars, array.get(lo_bars, i))
                        array.push(prices, array.get(lo_prices, i))

                    result := PatternResult.new("desc_tri", bar_index, bars, prices, slmax, slmin, intmax, intmin, tgt, -1)

        result

// PATTERN DETECTION: FLAG
// Based on zeta-zetra flag.py - Parallel channel with same-direction slopes
// ═══════════════════════════════════════════════════════════════
f_detect_flag(int lookback, int left, int right) =>
    PatternResult result = na

    if not i_flag_en
        result
    else
        [hi_bars, hi_prices, lo_bars, lo_prices, _, _, _] = f_collect_pivots(lookback, left, right)

        bool enough_points = (array.size(hi_bars) >= i_flag_min_pts or array.size(lo_bars) >= i_flag_min_pts) and array.size(hi_bars) > 0 and array.size(lo_bars) > 0

        if enough_points
            // Monotonicity check: prices must be consistently moving in one direction
            bool mono_hi = true
            bool mono_lo = true

            if array.size(hi_prices) >= 2
                for i = 1 to array.size(hi_prices) - 1
                    if array.get(hi_prices, i) < array.get(hi_prices, i - 1) - syminfo.mintick * 10
                        mono_hi := false
                        break
            if not mono_hi
                // Check if consistently decreasing
                mono_hi := true
                for i = 1 to array.size(hi_prices) - 1
                    if array.get(hi_prices, i) > array.get(hi_prices, i - 1) + syminfo.mintick * 10
                        mono_hi := false
                        break

            if array.size(lo_prices) >= 2
                for i = 1 to array.size(lo_prices) - 1
                    if array.get(lo_prices, i) < array.get(lo_prices, i - 1) - syminfo.mintick * 10
                        mono_lo := false
                        break
            if not mono_lo
                mono_lo := true
                for i = 1 to array.size(lo_prices) - 1
                    if array.get(lo_prices, i) > array.get(lo_prices, i - 1) + syminfo.mintick * 10
                        mono_lo := false
                        break

            if mono_hi and mono_lo
                array<float> hi_bars_f = array.new<float>()
                for i = 0 to array.size(hi_bars) - 1
                    array.push(hi_bars_f, float(array.get(hi_bars, i)))
                [slmax, intmax, r2max] = f_linregress(hi_bars_f, hi_prices)

                array<float> lo_bars_f = array.new<float>()
                for i = 0 to array.size(lo_bars) - 1
                    array.push(lo_bars_f, float(array.get(lo_bars, i)))
                [slmin, intmin, r2min] = f_linregress(lo_bars_f, lo_prices)

                bool good_fit = r2max >= i_flag_r2 and r2min >= i_flag_r2
                // Both slopes must have the same sign (both positive or both negative)
                bool same_direction = (slmax > 0 and slmin > 0) or (slmax < 0 and slmin < 0)
                // Parallelism check
                bool parallel = math.abs(slmax) > 1e-10 ? (slmin / slmax >= i_flag_ratio_lo and slmin / slmax <= i_flag_ratio_hi) : false

                if good_fit and same_direction and parallel
                    bool is_bull_flag = slmax < 0 and slmin < 0  // Bearish flag = bull continuation
                    float channel_height = (intmax + slmax * bar_index) - (intmin + slmin * bar_index)
                    float tgt = is_bull_flag ? (intmax + slmax * bar_index) + channel_height : (intmin + slmin * bar_index) - channel_height

                    int[] bars = array.new<int>()
                    float[] prices = array.new<float>()
                    for i = 0 to array.size(hi_bars) - 1
                        array.push(bars, array.get(hi_bars, i))
                        array.push(prices, array.get(hi_prices, i))
                    for i = 0 to array.size(lo_bars) - 1
                        array.push(bars, array.get(lo_bars, i))
                        array.push(prices, array.get(lo_prices, i))

                    result := PatternResult.new("flag", bar_index, bars, prices, slmax, slmin, intmax, intmin, tgt, is_bull_flag ? 1 : -1)

        result

// PATTERN DETECTION: PENNANT
// Based on zeta-zetra pennant.py - Converging lines with symmetric slopes
// ═══════════════════════════════════════════════════════════════
f_detect_pennant(int lookback, int left, int right) =>
    PatternResult result = na

    if not i_pennant_en
        result
    else
        [hi_bars, hi_prices, lo_bars, lo_prices, _, _, _] = f_collect_pivots(lookback, left, right)

        bool enough_points = (array.size(hi_bars) >= i_pennant_min_pts or array.size(lo_bars) >= i_pennant_min_pts) and array.size(hi_bars) > 0 and array.size(lo_bars) > 0

        if enough_points
            array<float> hi_bars_f = array.new<float>()
            for i = 0 to array.size(hi_bars) - 1
                array.push(hi_bars_f, float(array.get(hi_bars, i)))
            [slmax, intmax, r2max] = f_linregress(hi_bars_f, hi_prices)

            array<float> lo_bars_f = array.new<float>()
            for i = 0 to array.size(lo_bars) - 1
                array.push(lo_bars_f, float(array.get(lo_bars, i)))
            [slmin, intmin, r2min] = f_linregress(lo_bars_f, lo_prices)

            bool good_fit = r2max >= i_pennant_r2 and r2min >= i_pennant_r2
            // Pennant: highs falling, lows rising (converging)
            bool converging = slmin > 0 and slmax < 0
            // Symmetric slopes
            bool symmetric = math.abs(slmin) > 1e-10 ? (math.abs(slmax / slmin) >= i_pennant_ratio_lo and math.abs(slmax / slmin) <= i_pennant_ratio_hi) : false

            if good_fit and converging and symmetric
                float apex_x = math.abs(slmax - slmin) > 1e-10 ? (intmin - intmax) / (slmax - slmin) : float(bar_index + 50)
                float mid_price = (intmax + slmax * bar_index + intmin + slmin * bar_index) / 2
                float channel_height = (intmax + slmax * float(array.get(hi_bars, 0))) - (intmin + slmin * float(array.get(lo_bars, 0)))
                // Direction based on prior trend (price vs midline)
                bool is_bullish = close > mid_price
                float tgt = is_bullish ? close + channel_height * 0.618 : close - channel_height * 0.618

                int[] bars = array.new<int>()
                float[] prices = array.new<float>()
                for i = 0 to array.size(hi_bars) - 1
                    array.push(bars, array.get(hi_bars, i))
                    array.push(prices, array.get(hi_prices, i))
                for i = 0 to array.size(lo_bars) - 1
                    array.push(bars, array.get(lo_bars, i))
                    array.push(prices, array.get(lo_prices, i))

                result := PatternResult.new("pennant", bar_index, bars, prices, slmax, slmin, intmax, intmin, tgt, is_bullish ? 1 : -1)

        result

    string name = f_get_pattern_name(p.ptype)

    if array.size(p.bar_indices) >= 2 and array.size(p.prices) >= 2
        // Draw lines connecting the points
        for i = 0 to array.size(p.bar_indices) - 2
            int x1 = array.get(p.bar_indices, i)
            int x2 = array.get(p.bar_indices, i + 1)
            float y1 = array.get(p.prices, i)
            float y2 = array.get(p.prices, i + 1)
            pattern_lines.push(line.new(x1, y1, x2, y2, color=col, width=i_line_width, style=line.style_solid))

        // Label at the center
        int mid_bar = array.get(p.bar_indices, int(array.size(p.bar_indices) / 2))

        bool label_above = p.ptype == "double_top" or p.ptype == "hs"
        string tgt_str = not na(p.target) ? "\nTP: " + str.tostring(p.target, format.mintick) : ""
        pattern_labels.push(label.new(mid_bar, mid_price, name + tgt_str, style=label_above ? label.style_label_down : label.style_label_up, color=color.new(col, 85), textcolor=col, size=f_label_size()))

f_draw_regression_pattern(PatternResult p) =>
    color col = f_get_pattern_color(p.ptype)
    string name = f_get_pattern_name(p.ptype)

    if not na(p.sl_upper) and not na(p.sl_lower)
        // Determine x range from bar_indices
        int first_bar = bar_index
        int last_bar = bar_index
        if array.size(p.bar_indices) > 0
            first_bar := array.get(p.bar_indices, 0)
            last_bar := array.get(p.bar_indices, array.size(p.bar_indices) - 1)

        // Upper trendline
        float y1_upper = p.sl_upper * first_bar + p.int_upper
        float y2_upper = p.sl_upper * last_bar + p.int_upper
        pattern_lines.push(line.new(first_bar, y1_upper, last_bar, y2_upper, color=col, width=i_line_width, style=line.style_solid))

        // Lower trendline
        float y1_lower = p.sl_lower * first_bar + p.int_lower
        float y2_lower = p.sl_lower * last_bar + p.int_lower
        pattern_lines.push(line.new(first_bar, y1_lower, last_bar, y2_lower, color=col, width=i_line_width, style=line.style_solid))

        int mid_bar = int((first_bar + last_bar) / 2)
        float mid_price = (y1_upper + y2_lower) / 2
        string tgt_str = not na(p.target) ? "\nTP: " + str.tostring(p.target, format.mintick) : ""

var string last_alert_type = ""
var int last_alert_bar = 0

// Simple per-bar detection for alerts
int simple_pt = f_pivot_type(i_pivot_right, i_pivot_left, i_pivot_right)
bool new_pivot = simple_pt != 0

if barstate.isconfirmed and new_pivot and bar_index != last_alert_bar
    // Quick check for common patterns
    if i_alert_double
        PatternResult d = f_detect_double(i_lookback, i_pivot_left, i_pivot_right)
        if not na(d)
            if d.ptype == "double_top" and last_alert_type != "double_top"
                alert("🔴 Double Top detected on " + syminfo.ticker + " | Target: " + str.tostring(d.target, format.mintick), alert.freq_once_per_bar)
                last_alert_type := "double_top"
                last_alert_bar := bar_index
            if d.ptype == "double_bottom" and last_alert_type != "double_bottom"
                alert("🟢 Double Bottom detected on " + syminfo.ticker + " | Target: " + str.tostring(d.target, format.mintick), alert.freq_once_per_bar)
                last_alert_type := "double_bottom"
                last_alert_bar := bar_index

    if i_alert_triangle
        PatternResult t = f_detect_triangle(i_lookback, i_pivot_left, i_pivot_right)
        if not na(t)
            string tri_name = f_get_pattern_name(t.ptype)
            if t.ptype != last_alert_type
                string dir = t.direction == 0 ? "" : t.direction == 1 ? " Bullish" : " Bearish"
                alert(tri_name + dir + " detected on " + syminfo.ticker + " | Target: " + str.tostring(t.target, format.mintick), alert.freq_once_per_bar)
                last_alert_type := t.ptype
                last_alert_bar := bar_index

    if i_alert_flag
        PatternResult f = f_detect_flag(i_lookback, i_pivot_left, i_pivot_right)
        if not na(f) and last_alert_type != "flag"
            string dir = f.direction == 0 ? "" : f.direction == 1 ? " Bull" : " Bear"
            alert("🏳 Flag" + dir + " detected on " + syminfo.ticker + " | Target: " + str.tostring(f.target, format.mintick), alert.freq_once_per_bar)
            last_alert_type := "flag"
            last_alert_bar := bar_index

    if i_alert_pennant
        PatternResult pn = f_detect_pennant(math.min(i_lookback, 25), i_pivot_left, i_pivot_right)
        if not na(pn) and last_alert_type != "pennant"
            string dir = pn.direction == 0 ? "" : pn.direction == 1 ? " Bullish" : " Bearish"
            alert("🔻 Pennant" + dir + " detected on " + syminfo.ticker + " | Target: " + str.tostring(pn.target, format.mintick), alert.freq_once_per_bar)
            last_alert_type := "pennant"
            last_alert_bar := bar_index
